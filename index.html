Synchronisation Google Sheets

Echange de données bi-directionnel en temps réel

Entrez l'URL générée par votre script Google Sheets déployé pour activer la synchro bi-directionnelle.

Actions de Synchronisation Dernière synchronisation : Jamais
  1. Ouvrez votre feuille Google Sheet contenant vos onglets.
  2. Allez dans le menu Extensions ➔ Apps Script.
  3. Effacez le code existant et collez le script ci-dessous : 
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var keepsSheet = ss.getSheetByName("BottleKeep") || ss.getSheets()[0];
  var data = keepsSheet.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify({ status: "success", data: data }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("BottleKeep") || ss.getSheets()[0];
    
    if (contents.action === "push" && contents.bottleKeeps) {
      // Met à jour les lignes avec les données envoyées
      sheet.clearContents();
      sheet.appendRow(["ID", "BottleKeepNo", "Title", "BottleCatalog", "SellDate", "ExpirationDate", "LastUsageDate", "Status"]);
      contents.bottleKeeps.forEach(function(b) {
        sheet.appendRow([b.ID, b.BottleKeepNo, b.Title, b.BottleCatalog, b.SellDate, b.ExpirationDate, b.LastUsageDate, b.Status]);
      });
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
